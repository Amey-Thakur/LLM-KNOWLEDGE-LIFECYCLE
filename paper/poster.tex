% \resizebox{\linewidth}{!}{...}. That fixes the width exactly, but it also
% scales the type inside, so a figure whose natural width is far from its
% container renders its labels at a different size from the body text. Earlier
% drafts read as a page assembled from several documents for that reason. Each
% figure below is therefore drawn at a natural width of (container / 1.4), so

% leaves white down the right side of the block, which reads as a mistake rather
% than as air. The lane plans below are set so that the rightmost element of
% each figure lands on the right edge of its natural width.
%
% NO OVERLAP. Free text placed at hand-picked tikz coordinates is where words
% collide, because the width of a label is not known when its neighbor is
% positioned. Two defenses are used. Anything that is really a list, a callout
% or a code sample is built from LaTeX boxes, which cannot overlap by
% construction. Where geometry genuinely is needed, every text node is given an
% explicit text width, and the clearance to its neighbor is worked out from the
% average character advance of about 0.45 units at body size. The rendered PDF
% is then checked for intersecting text boxes, which is the only way to know.
%

% The identity band. It opens the page and closes it at full weight, and divides
% the header from the content at a third of that weight. Three strips of equal
% thickness would read as a repeating pattern rather than as a frame with a
% rule inside it, so the divider is deliberately thin.
%
% Three rules learned from proofs that came out wrong.
%
% The band is drawn 58 units wide and scaled to a width passed in by the caller.
% An earlier version targeted \textwidth, which is the 84.1 cm paper width, not
% the 82.5 cm measure, so both bands ran off the right edge of the sheet.
%
% Every coordinate is a literal. Written as \i*11.6 + 5.8, the arithmetic
% resolved for the fills but not for the labels, which landed on y = 0 instead
% of y = 0.95 and so straddled the bottom edge of their own band, half of each
% word in white on white.
%
% The bounding box is stated outright. Without it one stray node stretched a
% picture that should have been 58 units wide to 126, and since \resizebox fixes
% the width of the whole box, the visible part came out at 46 percent size in a
% band that was nominally full width.
% -----------------------------------------------------------------------------
\newcommand{\stageband}[2]{%  #1 = target width, #2 = height in band units
  \resizebox{#1}{!}{%
    \begin{tikzpicture}
      \foreach \c/\a/\b in {acquire/0/11.6, store/11.6/23.2, retrieve/23.2/34.8,
                            update/34.8/46.4, forget/46.4/58}
        {\fill[\c] (\a, 0) rectangle (\b, #2);}
      \useasboundingbox (0, 0) rectangle (58, #2);
    \end{tikzpicture}}%
}

% The three bands have to be the same width and sit on the same axis, but they
% are drawn from two different boxes: the title, whose measure is 82.5 cm, and
% the footer block, whose measure is 80.9 cm and which starts 0.8 cm further in.
% Left alone, the closing band would be 1.6 cm narrower than the opening one.
% \bandw is captured once from the title and every band is drawn at that width,
% shifted by half the difference so all three land on the same page axis. The
% enclosing \makebox is exactly the local measure, so the overhang does not
% report as an overfull line.
\newlength{\bandw}
% A fallback, so that a band is never asked to scale itself to zero width if the
% title is ever moved after something that draws one.
\AtBeginDocument{\ifdim\bandw=0pt \global\bandw=\textwidth\fi}
\newcommand{\atbandwidth}[1]{%
  \noindent\makebox[\linewidth][l]{%
    \hspace*{\dimexpr(\linewidth-\bandw)/2\relax}#1}%
}

% -----------------------------------------------------------------------------
% A callout: an accent rule, a tinted panel, and text that wraps inside it.
% Built from a self-sizing node rather than fixed coordinates, so the panel is
% always exactly as tall as its content and the text can never spill out of it.
% The 10 mm inner padding clears the 6 mm accent rule by 4 mm.
% -----------------------------------------------------------------------------
\newlength{\calloutwidth}
\newcommand{\callout}[2]{%  #1 = accent color, #2 = content
  \setlength{\calloutwidth}{\dimexpr\linewidth-20mm\relax}%
  \noindent\begin{tikzpicture}
    \node[fill=panel, text width=\calloutwidth, inner sep=10mm,
          align=left, anchor=north west] (c) at (0,0) {#2};
    \fill[#1] (c.south west) rectangle ([xshift=6mm]c.north west);
  \end{tikzpicture}%
}

\block{A metric}{\bodyset
  \textbf{Lifecycle Desynchronization} is the surprisal of the \textbf{correct}
  answer while the corrective document is present:
  \vspace{4mm}
  \begin{equation*}
    \widehat{\mathcal{D}}_{sync}
      = -\ln P\!\left(y^{*} \mid q,\, \theta_{stale},\, \mathcal{M}_{now}\right)
  \end{equation*}
  \vspace{2mm}

  It is read in nats: $n$ nats means probability $e^{-n}$. Past \textbf{9.2} the
  correct answer is below one chance in ten thousand.
  \vspace{8mm}

  % Natural width 18.2 units against a 25.5 cm column: the 1.4x figure scale.
  % 1.3 units per nat over a 0 to 14 range. Tick labels are about 1.1 units wide
  % and the closest pair, 9.2 and the 12.05 marker, are 3.7 units apart.
  \resizebox{\linewidth}{!}{%
  \begin{tikzpicture}
    \fill[ok!30]     (0,     0) rectangle (0.91,  1.5);
    \fill[yellow!45] (0.91,  0) rectangle (5.98,  1.5);
    \fill[orange!45] (5.98,  0) rectangle (11.96, 1.5);
    \fill[update!45] (11.96, 0) rectangle (18.20, 1.5);

    \draw[ink, line width=4pt] (15.67, -0.4) -- (15.67, 1.9);
    \node[anchor=south east, text=muted, font=\small] at (18.2, 2.0) {measured};

    \node[text=muted, font=\small] at (0.91,  -1.0) {0.7};
    \node[text=muted, font=\small] at (5.98,  -1.0) {4.6};
    \node[text=muted, font=\small] at (11.96, -1.0) {9.2};
    \node[font=\bfseries]          at (15.67, -1.0) {12.05};
    \useasboundingbox (0, -1.8) rectangle (18.2, 3.2);
  \end{tikzpicture}}
  \vspace{8mm}

  A companion number, \textbf{context influence}, asks whether the document
  moved the model at all. Here it is \textbf{0.033} nats: the evidence is
  present, and \textbf{inert}. The notice is placed in the prompt by
  construction, so nothing was lost in retrieval.
}

\column{0.34}
\colorlet{framecolor}{store}
\block{An architecture}{\bodyset
  Storage records \textbf{what} is true but never \textbf{when} it was learned,
  so nothing lets the model prefer fresh evidence over a confident old memory.
  \vspace{8mm}

  % Natural width 18.2 units against a 25.5 cm column: the 1.4x figure scale.
  % Slots are 5.1 wide on a 6.55 pitch, so the gutter is 1.45 and every label is
  % held inside a 4.3 unit text width, well clear of it. The row spans the full
  % 18.2 so the figure finishes on the right edge of the column.
  \resizebox{\linewidth}{!}{%
  \begin{tikzpicture}[
    lcslot/.style={rounded corners=3pt, minimum width=5.1cm, minimum height=3.2cm,
                   align=center, draw=hairline, line width=2pt,
                   font=\small, text width=4.3cm}
  ]
    \node[lcslot] (k) at (2.55,  0) {key\\ $k_i$};
    \node[lcslot] (v) at (9.10,  0) {value\\ $v_i$};
    \node[lcslot, draw=acquire, fill=acquire!12, text=acquire] (p) at (15.65, 0)
      {\textbf{provenance}\\ $p_i$};

    \draw[-{Stealth[length=4mm]}, muted,   line width=2pt] (k) -- (v);
    \draw[-{Stealth[length=4mm]}, acquire, line width=3pt] (v) -- (p);
    \node[below=4mm of p, text=acquire, font=\small, text width=5.1cm,
          align=center] {what we add};
    \useasboundingbox (0, -3.4) rectangle (18.2, 1.8);
  \end{tikzpicture}}
  \vspace{8mm}

  The \textbf{Provenance Vector} attaches acquisition time to every memory slot.
  A gate reads it at inference and mutes facts that have gone stale:
  \vspace{3mm}
  \begin{equation*}
    S(\Delta\tau_i) = \sigma\!\left(\gamma - \beta\,\Delta\tau_i\right)
  \end{equation*}
  \vspace{3mm}

  \textbf{Weights are never modified}, so the specificity bottleneck that limits
  static editing does not apply. Under \textbf{0.02\%} extra parameters.
}
