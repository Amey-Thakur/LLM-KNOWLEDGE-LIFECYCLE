\begin{frame}{The Fragmentation Problem}
  \begin{itemize}
    \item Research on LLM knowledge is siloed into subfields:
    \begin{itemize}
      \item RAG, knowledge editing, continual learning, context engineering, agent memory.
    \end{itemize}
    \item \textbf{The problem:} each stage is optimized in isolation; the \textbf{boundaries} between them go unstudied.
    \item \textbf{The evidence:} across 20 representative works, median lifecycle coverage is 2 of 5 stages.
  \end{itemize}
\end{frame}

% --- SLIDE 3 ---
\begin{frame}{Quantifying the Failure: $\mathcal{D}_{sync}$}
  \begin{itemize}
    \item \textbf{Lifecycle Desynchronization:} how far the deployed system's output is from what a synchronized system would produce.
    \item For a fact with verified answer $y^*$, it reduces to a surprisal:
  \end{itemize}
  \begin{equation*}
    \widehat{\mathcal{D}}_{sync} = -\ln P\!\left(y^* \mid q,\, \theta_{stale},\, \mathcal{M}_{now}\right)
  \end{equation*}
  \begin{itemize}
    \item $n$ nats means the correct answer holds probability $e^{-n}$.
    \item Above $9.2$ nats: no realistic decoding recovers the correct answer.
    \item Companion diagnostic $\mathcal{I}_{ctx}$: how far the context moves the distribution at all.
  \end{itemize}
\end{frame}

% --- SLIDE 4 ---
\begin{frame}{Measurement: The Vioxx Probe on GPT-2}
  \begin{itemize}
    \item \textbf{Scenario:} Vioxx withdrawn in 2004 for cardiovascular risk; withdrawal notice placed in context.
    \item \textbf{Measured:} $P(\text{``safe''})$ rises from 37.5\% to 42.6\% \textit{with} the notice present.
    \item \textbf{Measured:} $P(\text{``withdrawn''}) = 5.9 \times 10^{-6}$, so $\widehat{\mathcal{D}}_{sync} = 12.05$ nats.
    \item \textbf{Measured:} the notice shifts the full distribution by only $0.03$ nats.
    \item Retrieval succeeded; conflict resolution failed. Deterministic, reproducible script.
  \end{itemize}
\end{frame}

% --- SLIDE 5 ---
\begin{frame}{Root Cause and Proposal: The Provenance Vector ($p$)}
  \begin{itemize}
    \item Parametric storage discards \textit{when} and \textit{from where} a fact was learned.
    \item \textbf{Proposal:} each FFN memory slot $(k_i, v_i)$ gains metadata $p_i$ encoding acquisition time $\tau_i$ and source reliability $s_i$.
    \item Metadata feeds a gate; it never enters the residual stream.
  \end{itemize}
  \begin{equation*}
    \text{FFN}_{steered}(x) = \sum_{i=1}^{d_m} S(\Delta\tau_i)\, f(x \cdot k_i)\, v_i
  \end{equation*}
\end{frame}

% --- SLIDE 6 ---
\begin{frame}{Provenance-Gated Inference}
  \begin{equation*}
    S(\Delta \tau_i) = \sigma(\gamma - \beta \cdot \Delta \tau_i)
  \end{equation*}
  \begin{itemize}
    \item Fresh facts pass ($S \approx 1$); stale facts are attenuated ($S \to 0$).
    \item Conflicts resolve at inference time; base weights are never modified.
    \item Under stated idealized assumptions, gating drives $\mathcal{D}_{sync}$ to zero (Proposition 1 in the paper).
    \item Cost: about 1M extra parameters on a 7B model, below $0.02\%$.
  \end{itemize}
\end{frame}

% --- SLIDE 7 ---
\begin{frame}{Conclusion and Roadmap}
  \begin{itemize}
    \item Knowledge conflicts are real, measurable, and localized to stage boundaries.
    \item $\mathcal{D}_{sync}$ makes the failure auditable; the Provenance Vector targets its root cause.
    \item \textbf{Next:} scale the probe into a benchmark across models; extend to text-image conflicts on open VLMs.
  \end{itemize}
